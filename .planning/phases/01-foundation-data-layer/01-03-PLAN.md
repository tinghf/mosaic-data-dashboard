REQUIRED_COLUMNS_X = {"country_iso3", ...}  # set per dataset; sourced from COLUMN_DISCOVERY.md

def load_x(country: str | None = None) -> pd.DataFrame:
    root = resolve_data_root()
    subdir = root / "SUBDIR" / ("granularity" if any)
    if not subdir.exists():
        logging.getLogger(__name__).warning("SUBDIR not found at %s — returning empty", subdir)
        return _empty_x()
    files = sorted(subdir.glob("*.csv"))
    if not files:
        return _empty_x()
    csv_path = files[0]  # or the canonical filename, see RESEARCH.md
    mtime = csv_path.stat().st_mtime
    df = _read_x_cached(str(csv_path), mtime)
    if country is not None:
        df = df[df["country_iso3"] == country]
        if df.empty:
            return _empty_x()
    return df.reset_index(drop=True)

@st.cache_data(show_spinner=False)
def _read_x_cached(path: str, mtime: float) -> pd.DataFrame:
    df = pd.read_csv(path, parse_dates=[<date_cols_from_COLUMN_DISCOVERY>])
    # Rename upstream columns to canonical names per D-06/D-07 (use rename table from COLUMN_DISCOVERY.md)
    df = df.rename(columns={<upstream>: <canonical>, ...})
    require_columns(df, REQUIRED_COLUMNS_X, dataset="SUBDIR/granularity")
    return df

def _empty_x() -> pd.DataFrame:
    return pd.DataFrame({c: pd.Series(dtype="object") for c in REQUIRED_COLUMNS_X})
